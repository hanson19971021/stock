import asyncio, json, websockets
import shioaji as sj

api = sj.Shioaji(simulation=False)
api.login(person_id="YOUR_PERSON_ID", passwd="YOUR_PASSWORD", contract_fields=True)

connected_clients = set()
WATCH_LIST = ["2013", "2221", "3527", "4721", "3141", "6104", "2603", "2609", "3035", "3443"]

async def broadcast(message):
    if connected_clients:
        await asyncio.gather(*[client.send(message) for client in connected_clients], return_exceptions=True)

@api.on_tick()
def quote_callback(exchange, tick):
    data = {
        "code": tick.code, "price": float(tick.close),
        "volume": int(tick.volume), "total_volume": int(tick.total_volume),
        "change_rate": float(tick.pct_chg) if hasattr(tick, 'pct_chg') else 0.0,
        "time": tick.datetime.strftime("%H:%M:%S")
    }
    asyncio.run(broadcast(json.dumps(data)))

async def ws_handler(websocket, path):
    connected_clients.add(websocket)
    try:
        async for message in websocket: pass
    finally:
        connected_clients.remove(websocket)

for code in WATCH_LIST:
    try:
        contract = api.Contracts.Stocks[code]
        api.quote.subscribe(contract, quote_type=sj.constant.QuoteType.Tick)
    except Exception as e: print(e)

start_server = websockets.serve(ws_handler, "localhost", 8765)
asyncio.get_event_loop().run_until_complete(start_server)
asyncio.get_event_loop().run_forever()
